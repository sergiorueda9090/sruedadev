# reporte de ventas
def generar_reporte(ventas):
    total = sum(v.monto for v in ventas)
    promedio = total / len(ventas)
    return {
        'total': total,
        'promedio': promedio
    }

SELECT p.id, c.nombre, p.total
FROM pedidos p
JOIN clientes c
  ON c.id = p.cliente_id
WHERE p.estado = 'pagado'
ORDER BY p.creado DESC;
